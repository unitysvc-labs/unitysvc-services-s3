# seller.secrets.txt — the labs seller's secrets for this repo's S3 gateway
#                       services (labs/s3-relay, labs/s3-proxy).
#
# Not an example — these are the seller-TEST values (the platform's minio test
# bucket at s3.unitysvc.dev), which are not real production credentials, so the
# file is committed. Each service's offering references these as customer_secrets;
# they must exist in the seller secret store so gateway / integration tests
# resolve them via the ops_customer → seller-secret fallback, and be present in
# the local environment so `usvc_seller specs run-tests` resolves
# ${ customer_secrets.NAME } when it renders the connectivity probe / code example.
#
# Upload (staging — uses UNITYSVC_SELLER_API_* from ~/.zshrc):
#     usvc_seller secrets upload seller.secrets.txt
#
# Production: set each NAME below as a GitHub repository variable (a secret for
# the access/secret keys); the upload-to-production workflow seeds them into the
# production secret store automatically. One-time repo setup, not per-deploy.
#
# Sourceable shell format — values use ${NAME:-default}: the real value comes
# from the environment (a shell export, or a GitHub Actions secret/variable in
# CI), else the mock default. Source for local runs: `set -a; . ./seller.secrets.txt`.

# ── S3 upstream — labs/s3-relay, labs/s3-proxy (byok channel) ──
# The customer's own S3-compatible bucket. Defaults target the platform's minio
# test bucket at s3.unitysvc.dev; the access/secret keys carry no committed
# default (they are real credentials) and must come from the environment / a CI
# secret — the minio test key on staging, the customer's own key in production.

# S3_RELAY_ENDPOINT — REQUIRED. S3-compatible endpoint URL (prod examples:
# https://s3.amazonaws.com, https://<account>.r2.cloudflarestorage.com,
# https://<project>.supabase.co/storage/v1/s3). Default: the platform test S3.
export S3_RELAY_ENDPOINT="${S3_RELAY_ENDPOINT:-https://s3.unitysvc.dev}"

# S3_RELAY_BUCKET — REQUIRED. Bucket name. Default: the shared test bucket.
export S3_RELAY_BUCKET="${S3_RELAY_BUCKET:-demo}"

# S3_RELAY_REGION — REQUIRED. Bucket region (offering default us-east-1).
export S3_RELAY_REGION="${S3_RELAY_REGION:-us-east-1}"

# S3_RELAY_ACCESS_KEY — REQUIRED. Access key id for the bucket. No committed
# default (real credential): export it locally or set a CI secret.
export S3_RELAY_ACCESS_KEY="${S3_RELAY_ACCESS_KEY:-}"

# S3_RELAY_SECRET_KEY — REQUIRED. Secret access key for the bucket. No committed
# default (real credential): export it locally or set a CI secret.
export S3_RELAY_SECRET_KEY="${S3_RELAY_SECRET_KEY:-}"
